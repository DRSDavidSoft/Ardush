cmake_minimum_required(VERSION 3.16)
project(Ardush VERSION 2.0.0 LANGUAGES C CXX)
enable_testing()
add_library(ardush INTERFACE)
target_include_directories(ardush INTERFACE Ardush/src)
target_compile_features(ardush INTERFACE c_std_99)
add_executable(ardush_checks tests/checks.c)
add_executable(ardush_view_checks tests/view_checks.cpp)
add_executable(ardush_telnet_checks tests/telnet_checks.c)
add_executable(ardush_compat_checks tests/checks.c)
target_compile_definitions(ardush_compat_checks PRIVATE ARDUSH_RAW_ECHO=0 ARDUSH_LEGACY_EXIT_BYTES=1 PS_TERMINAL_USE_LIBC)
add_executable(ardush_ascii_checks tests/view_checks.cpp)
target_compile_features(ardush_ascii_checks PRIVATE cxx_std_11)
target_compile_definitions(ardush_ascii_checks PRIVATE ARDUSH_ASCII_ONLY)
target_compile_features(ardush_view_checks PRIVATE cxx_std_11)
foreach(check ardush_checks ardush_view_checks ardush_telnet_checks ardush_compat_checks ardush_ascii_checks)
  target_link_libraries(${check} PRIVATE ardush)
  if(MSVC)
    target_compile_options(${check} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${check} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror -fstack-usage)
  endif()
  add_test(NAME ${check} COMMAND ${check})
endforeach()
add_test(NAME shared_source_hashes COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-portable-shell.cmake")
add_executable(ardush_batch host/batch.c)
target_link_libraries(ardush_batch PRIVATE ardush)
install(DIRECTORY Ardush/src/ DESTINATION include/ardush FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp")
