cmake_minimum_required(VERSION 3.16)
project(SerialConsole VERSION 2.3.0 LANGUAGES C CXX)
enable_testing()
add_library(portable_console INTERFACE)
target_include_directories(portable_console INTERFACE SerialConsole/src)
target_compile_features(portable_console INTERFACE c_std_99)
add_executable(portable_console_checks tests/checks.c)
add_executable(portable_console_view_checks tests/view_checks.cpp)
add_executable(portable_console_telnet_checks tests/telnet_checks.c)
add_executable(portable_console_compat_checks tests/checks.c)
target_compile_definitions(portable_console_compat_checks PRIVATE PS_CONSOLE_RAW_ECHO=0 PS_CONSOLE_LEGACY_EXIT_BYTES=1 PS_TERMINAL_USE_LIBC PS_CONSOLE_SUGGESTIONS=0)
add_executable(portable_console_reduced_checks tests/view_checks.cpp)
target_compile_features(portable_console_reduced_checks PRIVATE cxx_std_11)
target_compile_definitions(portable_console_reduced_checks PRIVATE PS_DEVICE_VIEW_TABLE=0 PS_DEVICE_VIEW_WIDGETS=0 PS_DEVICE_VIEW_DIALOGS=0 PS_DEVICE_VIEW_MATRIX=0 PS_STYLE_THEMES=0)
add_executable(portable_console_ascii_checks tests/view_checks.cpp)
target_compile_features(portable_console_ascii_checks PRIVATE cxx_std_11)
target_compile_definitions(portable_console_ascii_checks PRIVATE PS_CONSOLE_ASCII_ONLY)
target_compile_features(portable_console_view_checks PRIVATE cxx_std_11)
foreach(check portable_console_checks portable_console_view_checks portable_console_telnet_checks portable_console_compat_checks portable_console_ascii_checks portable_console_reduced_checks)
  target_link_libraries(${check} PRIVATE portable_console)
  if(MSVC)
    target_compile_options(${check} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${check} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror -fstack-usage)
  endif()
  add_test(NAME ${check} COMMAND ${check})
endforeach()
add_test(NAME shared_source_hashes COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-portable-shell.cmake")
add_executable(portable_console_batch host/batch.c)
target_link_libraries(portable_console_batch PRIVATE portable_console)
install(DIRECTORY SerialConsole/src/ DESTINATION include/portable_console FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp")
foreach(mode CHECK UPDATE)
  string(TOLOWER "${mode}" target_mode)
  add_custom_target(previews_${target_mode}
    COMMAND "${CMAKE_COMMAND}" "-DMODE=${mode}" "-DBUILD_DIR=${CMAKE_BINARY_DIR}/previews"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/rebuild-previews.cmake"
    USES_TERMINAL VERBATIM)
endforeach()
