cmake_minimum_required(VERSION 3.20)
project(ConsolePreviews LANGUAGES C CXX)
if(NOT EXISTS "${REPLAY_SOURCE}")
  message(FATAL_ERROR "Run tools/rebuild-previews.cmake")
endif()
get_filename_component(console_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(console_preview "${console_root}/tests/render.cpp" "${REPLAY_SOURCE}")
target_include_directories(console_preview PRIVATE "${console_root}/SerialConsole/src")
target_compile_definitions(console_preview PRIVATE PS_REPLAY_ONLY)
target_compile_features(console_preview PRIVATE c_std_99 cxx_std_11)
set_target_properties(console_preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(console_preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(MSVC)
  # Keep portable ISO C stdio in this host-only fixture; fopen errors are checked.
  target_compile_definitions(console_preview PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(console_preview PRIVATE /W4 /WX)
else()
  target_compile_options(console_preview PRIVATE -Wall -Wextra -Werror)
endif()
