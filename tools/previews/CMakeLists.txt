cmake_minimum_required(VERSION 3.20)
project(ConsolePreviews LANGUAGES C CXX)
if(NOT EXISTS "${REPLAY_SOURCE}")
  message(FATAL_ERROR "Run tools/rebuild-previews.cmake")
endif()
get_filename_component(console_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(console_preview "${console_root}/tests/render.cpp" "${REPLAY_SOURCE}")
add_executable(full_preview "${console_root}/tests/render_profiles.cpp" "${REPLAY_SOURCE}")
add_executable(reduced_preview "${console_root}/tests/render_profiles.cpp" "${REPLAY_SOURCE}")
add_executable(prompt_preview "${console_root}/tests/render_prompt.c" "${REPLAY_SOURCE}")
target_compile_definitions(reduced_preview PRIVATE PS_PREVIEW_REDUCED=1 PS_DEVICE_VIEW_RETAIN_ROWS=0 PS_DEVICE_VIEW_WIDGETS=0
  PS_DEVICE_VIEW_DIALOGS=0 PS_DEVICE_VIEW_MATRIX=0 PS_STYLE_THEMES=0 PS_DEVICE_VIEW_VERTICAL_TABS=0)
target_compile_definitions(prompt_preview PRIVATE PS_CONSOLE_ENABLE_DEVICE_VIEW=0)
foreach(target console_preview full_preview reduced_preview prompt_preview)
target_include_directories(${target} PRIVATE "${console_root}/SerialConsole/src")
target_compile_definitions(${target} PRIVATE PS_REPLAY_ONLY)
target_compile_features(${target} PRIVATE c_std_99 cxx_std_11)
set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(MSVC)
  # Keep portable ISO C stdio in this host-only fixture; fopen errors are checked.
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(${target} PRIVATE /W4 /WX)
else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endif()
endforeach()
